    51→function SparkLine({ data, color, className }: { data: number[]; color: string; className?: string }) {
    52→  const max = Math.max(...data, 1);
    53→  const min = Math.min(...data, 0);
    54→  const range = max - min || 1;
    55→  const w = 80, h = 24;
    56→  const points = data.map((v, i) => `${(i / (data.length - 1)) * w},${h - ((v - min) / range) * h}`).join(' ');
    57→  return (
    58→    <svg viewBox={`0 0 ${w} ${h}`} className={className} preserveAspectRatio="none">
    59→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" opacity="0.6" />
    60→    </svg>
    61→  );
    62→}

    99→function relativeTime(dateStr: string): string {
   100→  const now = Date.now();
   101→  const then = new Date(dateStr || '').getTime();
   102→  if (isNaN(then)) return '';
   103→  const diffMs = now - then;
   104→  const diffSec = Math.floor(diffMs / 1000);
   105→  if (diffSec < 60) return 'just now';
   106→  const diffMin = Math.floor(diffSec / 60);
   107→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`;
   108→  const diffHr = Math.floor(diffMin / 60);
   109→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`;
   110→  const diffDay = Math.floor(diffHr / 24);
   111→  if (diffDay < 30) return `${diffDay} day${diffDay > 1 ? 's' : ''} ago`;
   112→  const diffMo = Math.floor(diffDay / 30);
   113→  return `${diffMo} month${diffMo > 1 ? 's' : ''} ago`;
   114→}

   182→function SeverityTrend() {
   183→  const [data, setData] = useState<{severity: string; count: number}[]>([]);
   184→  const [loading, setLoading] = useState(true);
   185→
   186→  useEffect(() => {
   187→    fetch('/api/findings?status=OPEN&limit=100')
   188→      .then(r => r.json())
   189→      .then(d => {
   190→        const findings = d.findings || [];
   191→        const counts: Record<string, number> = {};
   192→        findings.forEach((f: Record<string, unknown>) => {
   193→          const s = String(f.severity);
   194→          counts[s] = (counts[s] || 0) + 1;
   195→        });
   196→        setData(Object.entries(counts).map(([severity, count]) => ({severity, count})));
   197→        setLoading(false);
   198→      })
   199→      .catch(() => setLoading(false));
   200→  }, []);
   201→
   202→  const maxCount = Math.max(...data.map(d => d.count), 1);
   203→  const severityOrder = ['CRITICAL', 'HIGH', 'MEDIUM', 'LOW'];
   204→  const sorted = [...data].sort((a, b) => severityOrder.indexOf(a.severity) - severityOrder.indexOf(b.severity));
   205→  const colors: Record<string, string> = { CRITICAL: 'bg-red-500', HIGH: 'bg-orange-500', MEDIUM: 'bg-yellow-500', LOW: 'bg-emerald-500' };
   206→  const textColors: Record<string, string> = { CRITICAL: 'text-red-400', HIGH: 'text-orange-400', MEDIUM: 'text-yellow-400', LOW: 'text-emerald-400' };
   207→
   208→  return (
   209→    <Card className="border-border/50 rounded-xl">
   210→      <CardHeader className="pb-3">
   211→        <div className="flex items-center justify-between">
   212→          <CardTitle className="text-sm font-semibold">Open Finding Severity</CardTitle>
   213→          <Badge variant="outline" className="text-[10px]">Live</Badge>
   214→        </div>
   215→      </CardHeader>
   216→      <CardContent className="pt-0">
   217→        {loading ? <div className="h-20 skeleton" /> : (
   218→          <div className="space-y-3">
   219→            {sorted.map(item => (
   220→              <div key={item.severity} className="flex items-center gap-3">
   221→                <span className={`text-xs font-medium w-16 ${textColors[item.severity] || 'text-muted-foreground'}`}>{item.severity}</span>
   222→                <div className="flex-1 h-2 rounded-full bg-secondary overflow-hidden">
   223→                  <div className={`h-full rounded-full animate-fill-bar ${colors[item.severity] || 'bg-muted-foreground'}`} style={{ width: `${(item.count / maxCount) * 100}%`, transition: 'width 0.6s ease-out' }} />
   224→                </div>
   225→                <span className="text-xs font-mono w-6 text-right text-secondary-bright">{item.count}</span>
   226→              </div>
   227→            ))}
   228→          </div>
   229→        )}
   230→      </CardContent>
   231→    </Card>
   232→  );
   233→}
