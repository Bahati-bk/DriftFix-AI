    67→function relativeTime(dateStr: string): string {
    68→  const now = Date.now();
    69→  const then = new Date(dateStr || '').getTime();
    70→  if (isNaN(then)) return '';
    71→  const diffMs = now - then;
    72→  const diffSec = Math.floor(diffMs / 1000);
    73→  if (diffSec < 60) return 'just now';
    74→  const diffMin = Math.floor(diffSec / 60);
    75→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`;
    76→  const diffHr = Math.floor(diffMin / 60);
    77→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`;
    78→  const diffDay = Math.floor(diffHr / 24);
    79→  if (diffDay < 30) return `${diffDay} day${diffDay > 1 ? 's' : ''} ago`;
    80→  const diffMo = Math.floor(diffDay / 30);
    81→  return `${diffMo} month${diffMo > 1 ? 's' : ''} ago`;
    82→}
