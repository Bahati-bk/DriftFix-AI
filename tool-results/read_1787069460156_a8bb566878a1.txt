    51→    51→function SparkLine({ data, color, className }: { data: number[]; color: string; className?: string }) {
    52→    52→  const max = Math.max(...data, 1);
    53→    53→  const min = Math.min(...data, 0);
    54→    54→  const range = max - min || 1;
    55→    55→  const w = 80, h = 24;
    56→    56→  const points = data.map((v, i) => `${(i / (data.length - 1)) * w},${h - ((v - min) / range) * h}`).join(' ');
    57→    57→  return (
    58→    58→    <svg viewBox={`0 0 ${w} ${h}`} className={className} preserveAspectRatio="none">
    59→    59→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" opacity="0.6" />
    60→    60→    </svg>
    61→    61→  );
    62→    62→}

    99→    99→function relativeTime(dateStr: string): string {
   100→   100→  const now = Date.now();
   101→   101→  const then = new Date(dateStr || '').getTime();
   102→   102→  if (isNaN(then)) return '';
   103→   103→  const diffMs = now - then;
   104→   104→  const diffSec = Math.floor(diffMs / 1000);
   105→   105→  if (diffSec < 60) return 'just now';
   106→   106→  const diffMin = Math.floor(diffSec / 60);
   107→   107→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`;
   108→   108→  const diffHr = Math.floor(diffMin / 60);
   109→   109→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`;
   110→   110→  const diffDay = Math.floor(diffHr / 24);
   111→   111→  if (diffDay < 30) return `${diffDay} day${diffDay > 1 ? 's' : ''} ago`;
   112→   112→  const diffMo = Math.floor(diffDay / 30);
   113→   113→  return `${diffMo} month${diffMo > 1 ? 's' : ''} ago`;
   114→   114→}
