    70→function relativeTime(dateStr: string): string {
    71→  const now = Date.now();
    72→  const then = new Date(dateStr || '').getTime();
    73→  if (isNaN(then)) return '';
    74→  const diffMs = now - then;
    75→  const diffSec = Math.floor(diffMs / 1000);
    76→  if (diffSec < 60) return 'just now';
    77→  const diffMin = Math.floor(diffSec / 60);
    78→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`;
    79→  const diffHr = Math.floor(diffMin / 60);
    80→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`;
    81→  const diffDay = Math.floor(diffHr / 24);
    82→  if (diffDay < 30) return `${diffDay} day${diffDay > 1 ? 's' : ''} ago`;
    83→  const diffMo = Math.floor(diffDay / 30);
    84→  return `${diffMo} month${diffMo > 1 ? 's' : ''} ago`;
    85→}
