    82→function relativeTime(dateStr: string): string {
    83→  const now = Date.now();
    84→  const then = new Date(dateStr || '').getTime();
    85→  if (isNaN(then)) return '';
    86→  const diffMs = now - then;
    87→  const diffSec = Math.floor(diffMs / 1000);
    88→  if (diffSec < 60) return 'just now';
    89→  const diffMin = Math.floor(diffSec / 60);
    90→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`;
    91→  const diffHr = Math.floor(diffMin / 60);
    92→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`;
    93→  const diffDay = Math.floor(diffHr / 24);
    94→  if (diffDay < 30) return `${diffDay} day${diffDay > 1 ? 's' : ''} ago`;
    95→  const diffMo = Math.floor(diffDay / 30);
    96→  return `${diffMo} month${diffMo > 1 ? 's' : ''} ago`;
    97→}
