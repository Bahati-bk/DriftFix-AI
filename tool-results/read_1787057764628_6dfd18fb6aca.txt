    66→function relativeTime(dateStr: string): string {
    67→  const now = Date.now();
    68→  const then = new Date(dateStr || '').getTime();
    69→  if (isNaN(then)) return '';
    70→  const diffMs = now - then;
    71→  const diffSec = Math.floor(diffMs / 1000);
    72→  if (diffSec < 60) return 'just now';
    73→  const diffMin = Math.floor(diffSec / 60);
    74→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`;
    75→  const diffHr = Math.floor(diffMin / 60);
    76→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`;
    77→  const diffDay = Math.floor(diffHr / 24);
    78→  if (diffDay < 30) return `${diffDay} day${diffDay > 1 ? 's' : ''} ago`;
    79→  const diffMo = Math.floor(diffDay / 30);
    80→  return `${diffMo} month${diffMo > 1 ? 's' : ''} ago`;
    81→}
