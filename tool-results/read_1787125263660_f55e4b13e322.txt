    98→  if (!str) return '—';

   102→// ──────────────────────────────────────────────
   103→// QR Code Data URI (simple deterministic SVG-based marker, not a true QR encoder)
   104→// Generates a recognizable tamper-evident badge containing the short hash
   105→// ──────────────────────────────────────────────
   106→
   107→function integrityBadgeSVG(shortHash: string, timestamp: string): string {
   108→  // Create a deterministic visual fingerprint using the shortHash chars
   109→  const cells: string[] = [];
   110→  const hashChars = (shortHash + timestamp).split('');
   111→  const size = 9;
   112→  for (let y = 0; y < size; y++) {
   113→    for (let x = 0; x < size; x++) {
   114→      const idx = (y * size + x) % hashChars.length;
   115→      const charCode = hashChars[idx].charCodeAt(0);
   116→      // Finder pattern corners
   117→      const isFinder = (x < 3 && y < 3) || (x >= size - 3 && y < 3) || (x < 3 && y >= size - 3);
   118→      const filled = isFinder || (charCode % 3 !== 0);
   119→      cells.push(`<rect x="${x * 4}" y="${y * 4}" width="4" height="4" fill="${filled ? '#2dd4bf' : 'rgba(148,163,184,0.08)'}" rx="0.5"/>`);
   120→    }
   121→  }
   122→  return `
   123→    <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 36 36" width="36" height="36">
   124→      <rect width="36" height="36" fill="#0a0f1a" rx="4"/>
   125→      ${cells.join('')}
   126→    </svg>`;
   127→}
   128→
   129→function integrityBadgeDataUri(shortHash: string, timestamp: string): string {
   130→  const svg = integrityBadgeSVG(shortHash, timestamp)
   131→    .replace(/#/g, '%23')
   132→    .replace(/</g, '%3C')
   133→    .replace(/>/g, '%3E')
   134→    .replace(/"/g, "'");
   135→  return `data:image/svg+xml,${svg}`;
   136→}
