    45→    99→function relativeTime(dateStr: string): string {
    46→   100→  const now = Date.now();
    47→   101→  const then = new Date(dateStr || '').getTime();
    48→   102→  if (isNaN(then)) return '';
    49→   103→  const diffMs = now - then;
    50→   104→  const diffSec = Math.floor(diffMs / 1000);
    51→   105→  if (diffSec < 60) return 'just now';
    52→   106→  const diffMin = Math.floor(diffSec / 60);
    53→   107→  if (diffMin < 60) return `${diffMin} minute${diffMin > 1 ? 's' : ''} ago`;
    54→   108→  const diffHr = Math.floor(diffMin / 60);
    55→   109→  if (diffHr < 24) return `${diffHr} hour${diffHr > 1 ? 's' : ''} ago`;
    56→   110→  const diffDay = Math.floor(diffHr / 24);
    57→   111→  if (diffDay < 30) return `${diffDay} day${diffDay > 1 ? 's' : ''} ago`;
    58→   112→  const diffMo = Math.floor(diffDay / 30);
    59→   113→  return `${diffMo} month${diffMo > 1 ? 's' : ''} ago`;
    60→   114→}
